window.startGame = function(type) { document.getElementById('start').style.display = 'none'; const filename = type === 'game' ? 'Game-Theme.mp3' : 'Phonk-Theme.mp3'; currentMusic = new Audio(`assets/${filename}`); currentMusic.volume = 0.5; currentMusic.loop = true; currentMusic.play().catch(err => { console.error("Music failed:", err); alert("Music blocked? Try clicking again or check filename in assets/"); }); gameStarted = true; controls.enabled = false; };
